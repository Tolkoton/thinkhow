import os, sys
from typing import Literal
from pydantic import create_model
from thinkhow.facilitation.types import FacilitatorUpdate
from thinkhow.platform.config import load_settings
from thinkhow.platform.llm import make_llm, Turn, ReplyDelta
P = Literal["phase_1_time_travel", "phase_2_blind_spots"] | None
S = create_model("FacilitatorAnswer", __base__=FacilitatorUpdate, current_phase_id=(P, ...))
S.model_fields  # noqa
print("schema:", S.model_json_schema()["properties"]["current_phase_id"])
env = {**os.environ, "THINKHOW_LOCAL": "false", "THINKHOW_USERS_BUCKET": "-", "THINKHOW_SYSTEM_BUCKET": "-", "THINKHOW_OWNER_EMAIL": "-"}
llm = make_llm(load_settings(env), script=lambda *_: "")
sysmsg = ("Ти фасилітатор. Якщо в людини немає плану, current_phase_id = null; якщо є план — phase_1_time_travel. "
          "reply — одне коротке питання. phase_exit_met=false, map_updates=[]")
for text in ["Плану ще немає, шукаю, як заробити.", "Мій план: за 3 місяці відкрити кав'ярню у Львові."]:
    raw, n = None, 0
    for ev in llm.stream_json(sysmsg, [Turn("user", text)], S, None):
        if isinstance(ev, ReplyDelta): n += 1
        else: raw = ev.raw_json
    v = S.model_validate_json(raw)
    print(repr(text), "-> deltas", n, "phase", v.current_phase_id, "| reply:", v.reply)
